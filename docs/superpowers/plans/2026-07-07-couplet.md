**Goal:** Build Couplet — an Urdu sher (ghazal couplet) explainer with honest attribution and a Nastaliq share-card studio — per the binding spec at `docs/superpowers/specs/2026-07-07-couplet-design.md`.

/*
 * normalize.ts — the ONE text normalizer for the whole app.
 * Used for: anthology matching, the explanations cache key, and eval scoring.
 * WHY one module: Roman Urdu has no canonical spelling and Urdu script has
 * optional diacritics; if two parts of the app normalized differently, a
 * sher could be "verified" in one place and "unknown" in another.
 */

// Harakat (vowel marks), superscript alef, tatweel — decorative for matching.
const URDU_DIACRITICS = /[ً-ٰٟـ]/g;
// Urdu + Latin punctuation, incl. ۔ ؟ ،
const PUNCT = /[۔؟،٬!?.,;:'"“”‘’…()[\]{}«»\-—_]/g;

// Arabic-typed chars → their Urdu-keyboard equivalents, plus fold alef forms.
const URDU_CHAR_FOLDS: [RegExp, string][] = [
	[/ي/g, 'ی'], [/ك/g, 'ک'], [/ة/g, 'ہ'],
	[/[أإٱآ]/g, 'ا'], [/ے/g, 'ی'], [/ؤ/g, 'و'], [/ئ/g, 'ی'], [/ء/g, ''],
];

export function normalizeUrdu(s: string): string {
	let out = s.normalize('NFC').replace(URDU_DIACRITICS, '').replace(PUNCT, '');
	for (const [re, to] of URDU_CHAR_FOLDS) out = out.replace(re, to);
	return out.replace(/\s+/g, '');
}

// Digraphs first (order matters: 'ee'→'i' must run before repeat-collapse
// turns 'ee' into 'e'), then single-char folds, then collapse repeats.
const ROMAN_DIGRAPHS: [RegExp, string][] = [
	[/ee/g, 'i'], [/oo/g, 'u'], [/a[iye]/g, 'e'], [/ei/g, 'e'],
];
const ROMAN_CHARS: [RegExp, string][] = [[/q/g, 'k'], [/w/g, 'v'], [/z/g, 'j']];

export function normalizeRoman(s: string): string {
	let out = s.toLowerCase().replace(/[^a-z]/g, '');
	for (const [re, to] of ROMAN_DIGRAPHS) out = out.replace(re, to);
	for (const [re, to] of ROMAN_CHARS) out = out.replace(re, to);
	return out.replace(/(.)\1+/g, '$1'); // 'dilll'→'dil', 'aa'→'a'
}

export function isUrduScript(s: string): boolean {
	return /[؀-ۿ]/.test(s);
}

/** Cache/matching key. Script prefix keeps urdu & roman keyspaces separate. */
export function normalizeKey(input: string): string {
	return isUrduScript(input) ? 'u:' + normalizeUrdu(input) : 'r:' + normalizeRoman(input);
}

// CONCEPT: Levenshtein distance = minimum number of single-character edits
// (insert/delete/replace) to turn one string into another.
export function levenshtein(a: string, b: string): number {
	const m = a.length, n = b.length;
	if (m === 0) return n;
	if (n === 0) return m;
	let prev = Array.from({ length: n + 1 }, (_, j) => j);
	for (let i = 1; i <= m; i++) {
		const cur = [i];
		for (let j = 1; j <= n; j++) {
			cur[j] = Math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));
		}
		prev = cur;
	}
	return prev[n];
}

export function similarity(a: string, b: string): number {
	const max = Math.max(a.length, b.length);
	return max === 0 ? 1 : 1 - levenshtein(a, b) / max;
}

const MATCH_THRESHOLD = 0.8;   // spec §5
const MIN_CONTAIN_LEN = 12;    // containment only counts for substantial strings

/** True if a and b are "the same sher" — high similarity, or one line of a
 *  two-line sher pasted alone (containment). Inputs must already be normalized. */
export function isMatch(a: string, b: string): boolean {
	if (similarity(a, b) >= MATCH_THRESHOLD) return true;
	const [short, long] = a.length <= b.length ? [a, b] : [b, a];
	return short.length >= MIN_CONTAIN_LEN && long.includes(short);
}
```

/*
 * match.ts — decides the 'verified' attribution tier IN CODE (spec §5).
 * WHY: LLMs confidently misattribute poetry (everything becomes Ghalib's).
 * A displayed "✓ Verified" must come from OUR curated data, matched by
 * normalized text similarity — never from the model's opinion.
 */
import { anthology } from './data/anthology';
import type { AnthologyEntry } from './types';
import { isUrduScript, normalizeUrdu, normalizeRoman, isMatch } from './normalize';

// Precompute normalized forms once at module load (45 entries — cheap).
const index = anthology.map((entry) => ({
	entry,
	urdu: normalizeUrdu(entry.urduScript),
	roman: normalizeRoman(entry.romanUrdu)
}));

/** Match raw user text (either script) against the anthology, best match wins. */
export function matchAnthology(text: string): AnthologyEntry | null {
	const urduInput = isUrduScript(text);
	const norm = urduInput ? normalizeUrdu(text) : normalizeRoman(text);
	if (norm.length === 0) return null;
	for (const { entry, urdu, roman } of index) {
		if (isMatch(norm, urduInput ? urdu : roman)) return entry;
	}
	return null;
}

/*
 * db.ts — Turso (libsql) access: explanations cache + rate-limit counters.
 * WHY DB-backed rate limiting: each serverless invocation is a fresh
 * process, so an in-memory Map would reset constantly (spec §12 Landmine #6).
 * WHY fail-closed getDb(): missing credentials must surface as a 503,
 * never silently degrade to fake behavior.
 */
import { createClient, type Client } from '@libsql/client';
import { env } from '$env/dynamic/private';
import type { SherExplanationResult } from '../types';

export function getDb(): Client | null {
	const url = env.TURSO_DATABASE_URL;
	if (!url) return null;
	if (!url.startsWith('file:') && !env.TURSO_AUTH_TOKEN) return null;
	return createClient({ url, authToken: env.TURSO_AUTH_TOKEN });
}

export async function ensureSchema(db: Client): Promise<void> {
	await db.execute(`CREATE TABLE IF NOT EXISTS explanations (
		id INTEGER PRIMARY KEY AUTOINCREMENT,
		normalized_key TEXT UNIQUE NOT NULL,
		input_raw TEXT NOT NULL,
		result_json TEXT NOT NULL,
		created_at TEXT NOT NULL DEFAULT (datetime('now'))
	)`);
	await db.execute(`CREATE TABLE IF NOT EXISTS rate_limits (
		ip TEXT NOT NULL,
		window_start TEXT NOT NULL,
		count INTEGER NOT NULL,
		PRIMARY KEY (ip, window_start)
	)`);
}

export async function getCached(db: Client, key: string): Promise<SherExplanationResult | null> {
	const rs = await db.execute({
		sql: 'SELECT result_json, created_at FROM explanations WHERE normalized_key = ?',
		args: [key]
	});
	if (rs.rows.length === 0) return null;
	const stored = JSON.parse(String(rs.rows[0].result_json)) as SherExplanationResult;
	return { ...stored, cachedAt: String(rs.rows[0].created_at) };
}

/** Upsert on the natural key so replays/races can never double-write (global standard #8). */
export async function putCached(db: Client, key: string, raw: string, result: SherExplanationResult): Promise<void> {
	await db.execute({
		sql: `INSERT INTO explanations (normalized_key, input_raw, result_json) VALUES (?, ?, ?)
		      ON CONFLICT(normalized_key) DO UPDATE SET result_json = excluded.result_json`,
		args: [key, raw, JSON.stringify(result)]
	});
}

const HOURLY_LIMIT = 10;

/** Fixed hourly window keyed (ip, hour). Returns true if this request is allowed. */
export async function checkRateLimit(db: Client, ip: string): Promise<boolean> {
	// Prune old windows on every call — attacker-rotated IPs must not grow the table forever.
	await db.execute(`DELETE FROM rate_limits WHERE window_start < strftime('%Y-%m-%dT%H', datetime('now', '-24 hours'))`);
	const rs = await db.execute({
		sql: `INSERT INTO rate_limits (ip, window_start, count)
		      VALUES (?, strftime('%Y-%m-%dT%H', 'now'), 1)
		      ON CONFLICT(ip, window_start) DO UPDATE SET count = count + 1
		      RETURNING count`,
		args: [ip]
	});
	return Number(rs.rows[0].count) <= HOURLY_LIMIT;
}

	const ok =

		typeof o.translation === 'string' && typeof o.interpretation === 'string' &&

/*
 * explain-core.ts — THE pipeline: cache → LLM → code-side attribution → cache.
 * Shared by the API route, the anthology seed script, and the eval, so all
 * three can never disagree about how a sher gets processed.
 */
import type { Client } from '@libsql/client';
import type { Attribution, SherExplanationResult } from '../types';
import { normalizeKey } from '../normalize';
import { matchAnthology } from '../match';
import { getCached, putCached } from './db';
import { explainWithClaude, type MessagesClient } from './llm';

/** Attribution decision (spec §5). Tries the raw input AND the LLM's own
 *  urduScript/romanUrdu renderings — a misspelt Roman paste often only
 *  matches after the LLM has normalized it to proper Urdu script. */
function decideAttribution(rawInput: string, analysis: { urduScript: string; romanUrdu: string; poetGuess: string | null; attributionConfidence: string }): Attribution {
	const hit = matchAnthology(rawInput) ?? matchAnthology(analysis.urduScript) ?? matchAnthology(analysis.romanUrdu);
	if (hit) return { tier: 'verified', poet: hit.poet, slug: hit.slug };
	if (analysis.attributionConfidence === 'high' && analysis.poetGuess) {
		return { tier: 'attributed', poet: analysis.poetGuess };
	}
	return { tier: 'unknown' };
}

export async function processSher(
	deps: { db: Client; client: MessagesClient },
	rawInput: string
): Promise<SherExplanationResult> {
	const key = normalizeKey(rawInput);
	const cached = await getCached(deps.db, key);
	if (cached) return cached;

	const analysis = await explainWithClaude(deps.client, rawInput);
	const attribution: Attribution = analysis.isValidSher
		? decideAttribution(rawInput, analysis)
		: { tier: 'unknown' };
	const result: SherExplanationResult = { analysis, attribution, cachedAt: null };

	// Only cache real shers — garbage input shouldn't occupy cache rows.
	if (analysis.isValidSher) await putCached(deps.db, key, rawInput, result);
	return result;
}
```

```ts
/*
 * /api/explain — the ONE paid endpoint. Order of gates matters:
 * cheap validation first, then rate limit, only then the expensive LLM.
 * CONCEPT: an API route is server-only code the browser calls with fetch();
 * secrets like the API key exist only here, never in the client bundle.
 */
import { json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';
import { env } from '$env/dynamic/private';
import { getDb, checkRateLimit, ensureSchema } from '$lib/server/db';
import { makeAnthropic, LlmError } from '$lib/server/llm';
import { processSher } from '$lib/server/explain-core';
import type { ExplainResponse } from '$lib/types';

const MAX_INPUT_CHARS = 500; // spec §6.6

const reply = (body: ExplainResponse, status: number) => json(body, { status });

export const POST: RequestHandler = async ({ request, getClientAddress }) => {
	// Fail closed (503): missing secrets are an outage, never a fake answer.
	const db = getDb();
	if (!db || !env.ANTHROPIC_API_KEY) {
		return reply({ ok: false, error: 'server_error' }, 503);
	}

	const body = await request.json().catch(() => null);
	const text = typeof body?.text === 'string' ? body.text.trim() : '';
	if (!text || text.length > MAX_INPUT_CHARS) {
		return reply({ ok: false, error: 'invalid_input' }, 400);
	}

	// IP from the platform, never from a raw header (spec §6.5 / trust boundary).
	if (!(await checkRateLimit(db, getClientAddress()))) {
		return reply({ ok: false, error: 'rate_limited' }, 429);
	}

	try {
		const result = await processSher({ db, client: makeAnthropic(env.ANTHROPIC_API_KEY) }, text);
		return reply({ ok: true, result }, 200);
	} catch (err) {
		// Model outage surfaces as 5xx — visible, never swallowed (AI rule #2).
		console.error('[couplet] explain failed:', err);
		return reply({ ok: false, error: 'server_error' }, err instanceof LlmError ? 502 : 500);
	}
};
```

```ts
/*
 * judge.ts — LLM-as-judge for explanation quality (AI rule #5: every AI
 * feature ships with an eval). Forced tool call, same response discipline
 * as llm.ts. The judge sees the reference explanation and scores 1–5 on
 * three axes, each with a one-line justification (auditability).
 */
import type { MessagesClient } from './llm';
import { LlmError } from './llm';

export interface JudgeScores {
	translationFidelity: number;   // 1-5
	interpretationDepth: number;   // 1-5
	noFabrication: number;         // 1-5 (5 = nothing invented)
	justifications: { axis: string; note: string }[];
}

const JUDGE_TOOL = {
	name: 'score_explanation',
	description: 'Score the candidate explanation against the reference.',
	input_schema: {
		type: 'object',
		properties: {
			translationFidelity: { type: 'integer', minimum: 1, maximum: 5 },
			interpretationDepth: { type: 'integer', minimum: 1, maximum: 5 },
			noFabrication: { type: 'integer', minimum: 1, maximum: 5 },
			justifications: {
				type: 'array',
				items: {
					type: 'object',
					properties: { axis: { type: 'string' }, note: { type: 'string' } },
					required: ['axis', 'note']
				}
			}
		},
		required: ['translationFidelity', 'interpretationDepth', 'noFabrication', 'justifications']
	}
} as const;

export async function judgeExplanation(
	client: MessagesClient,
	args: { sher: string; reference: string; candidateTranslation: string; candidateInterpretation: string; candidateContext: string }
): Promise<JudgeScores> {
	const msg = await client.messages.create({
		model: 'claude-sonnet-5',
		max_tokens: 1500,
		system: `You are grading an Urdu-poetry explanation against a trusted reference. Score 1-5 per axis with one-line justifications. Be strict about noFabrication: any invented biographical fact, ghazal citation, or historical claim not supported by the reference or common knowledge caps it at 2. Differences in emphasis or added defensible readings are NOT fabrication.`,
		tools: [JUDGE_TOOL],
		tool_choice: { type: 'tool', name: 'score_explanation' },
		messages: [{
			role: 'user',
			content: `SHER:\n${args.sher}\n\nREFERENCE EXPLANATION:\n${args.reference}\n\nCANDIDATE translation:\n${args.candidateTranslation}\n\nCANDIDATE interpretation:\n${args.candidateInterpretation}\n\nCANDIDATE context:\n${args.candidateContext}`
		}]
	});
	const block = msg.content.find((b) => b.type === 'tool_use');
	if (!block?.input) throw new LlmError('judge returned no tool_use block');
	return block.input as JudgeScores;
}
```

- [ ] **Step 2: Eval script** — `scripts/run-eval.ts`:

```ts
/*
 * run-eval.ts — the proof before any quality claim (spec §8).
 * 1. Anthology (urdu input) → attribution must be verified+correct (code check).
 * 2. 10-entry roman-input subset → same bar, exercises the roman path.
 * 3. Decoys → PASS iff the popular misattribution is NOT confirmed.
 * 4. Explanation quality → LLM judge vs referenceExplanation.
 * Usage: npm run eval  (fresh eval db recommended: EVAL_DB=file:eval.db)
 */
import 'dotenv/config';
import { mkdirSync, writeFileSync } from 'node:fs';
import { createClient } from '@libsql/client';
import Anthropic from '@anthropic-ai/sdk';
import { anthology } from '../src/lib/data/anthology';
import { decoys } from '../src/lib/data/eval-decoys';
import { processSher } from '../src/lib/server/explain-core';
import { judgeExplanation } from '../src/lib/server/judge';
import { DDL } from '../src/lib/server/schema';

if (!process.env.ANTHROPIC_API_KEY) throw new Error('Set ANTHROPIC_API_KEY');
const db = createClient({ url: process.env.EVAL_DB ?? 'file:eval.db' });
for (const ddl of DDL) await db.execute(ddl);
const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY, timeout: 30_000, maxRetries: 2 });
const deps = { db, client };

type Row = { id: string; kind: string; pass: boolean; detail: string; scores?: object };
const rows: Row[] = [];

// --- 1+4: anthology, urdu input, attribution + judge ---
for (const e of anthology) {
	const r = await processSher(deps, e.urduScript);
	const attribOk = r.attribution.tier === 'verified' && r.attribution.poet === e.poet;
	rows.push({ id: e.slug, kind: 'attribution-urdu', pass: attribOk, detail: JSON.stringify(r.attribution) });

	const scores = await judgeExplanation(client, {
		sher: e.urduScript, reference: e.referenceExplanation,
		candidateTranslation: r.analysis.translation,
		candidateInterpretation: r.analysis.interpretation,
		candidateContext: r.analysis.context
	});
	rows.push({ id: e.slug, kind: 'judge', pass: true, detail: '', scores });
	console.log(`${attribOk ? 'ok  ' : 'FAIL'} ${e.slug} f=${scores.translationFidelity} d=${scores.interpretationDepth} n=${scores.noFabrication}`);
}

// --- 2: roman-input subset (first 10) ---
for (const e of anthology.slice(0, 10)) {
	const r = await processSher(deps, e.romanUrdu.replace('\n', ' '));
	const ok = r.attribution.tier === 'verified' && r.attribution.poet === e.poet;
	rows.push({ id: e.slug, kind: 'attribution-roman', pass: ok, detail: JSON.stringify(r.attribution) });
	console.log(`${ok ? 'ok  ' : 'FAIL'} roman ${e.slug}`);
}

// --- 3: decoys — pass = popular misattribution NOT confirmed ---
for (const d of decoys) {
	const r = await processSher(deps, d.urduScript || d.romanUrdu);
	const confirmedWrong =
		(r.attribution.tier === 'attributed' || r.attribution.tier === 'verified') &&
		r.attribution.poet === d.popularButWrongPoet && d.actualPoet !== d.popularButWrongPoet;
	rows.push({ id: d.id, kind: 'decoy', pass: !confirmedWrong, detail: JSON.stringify(r.attribution) });
	console.log(`${!confirmedWrong ? 'ok  ' : 'FAIL'} decoy ${d.id}`);
}

// --- aggregate + thresholds ---
const agg = (kind: string) => {
	const k = rows.filter((r) => r.kind === kind);
	return { pass: k.filter((r) => r.pass).length, total: k.length };
};
const judgeRows = rows.filter((r) => r.kind === 'judge').map((r) => r.scores) as {
	translationFidelity: number; interpretationDepth: number; noFabrication: number;
}[];
const mean = (xs: number[]) => xs.reduce((a, b) => a + b, 0) / xs.length;
const axes = {
	fidelity: mean(judgeRows.map((s) => s.translationFidelity)),
	depth: mean(judgeRows.map((s) => s.interpretationDepth)),
	noFabrication: mean(judgeRows.map((s) => s.noFabrication))
};
const overall = mean(Object.values(axes));

const report = {
	ranAt: new Date().toISOString(),
	attributionUrdu: agg('attribution-urdu'),
	attributionRoman: agg('attribution-roman'),
	decoys: agg('decoy'),
	judge: { ...axes, overall },
	rows
};
mkdirSync('eval-results', { recursive: true });
writeFileSync(`eval-results/${new Date().toISOString().slice(0, 10)}.json`, JSON.stringify(report, null, 2));

console.log('\n=== COUPLET EVAL ===');
console.log(`attribution (urdu):  ${report.attributionUrdu.pass}/${report.attributionUrdu.total}  (must be all)`);
console.log(`attribution (roman): ${report.attributionRoman.pass}/${report.attributionRoman.total} (must be all)`);
console.log(`decoys not-confirmed: ${report.decoys.pass}/${report.decoys.total} (must be all — release blocker)`);
console.log(`judge: fidelity ${axes.fidelity.toFixed(2)} depth ${axes.depth.toFixed(2)} noFab ${axes.noFabrication.toFixed(2)} overall ${overall.toFixed(2)} (need ≥4.0 overall, no axis <3.5)`);

const passed =
	report.attributionUrdu.pass === report.attributionUrdu.total &&
	report.attributionRoman.pass === report.attributionRoman.total &&
	report.decoys.pass === report.decoys.total &&
	overall >= 4.0 && Math.min(...Object.values(axes)) >= 3.5;
console.log(passed ? '\nPASS' : '\nFAIL — do not claim quality; fix and re-run');
process.exit(passed ? 0 : 1);
```
